const SHEET_ID = SpreadsheetApp.getActiveSpreadsheet().getId();

// ── CORS headers added to every response ──
function corsHeaders() {
  return {
    'Access-Control-Allow-Origin': '*',
    'Access-Control-Allow-Methods': 'POST, GET, OPTIONS',
    'Access-Control-Allow-Headers': 'Content-Type',
  };
}

function makeResponse(data) {
  const output = ContentService
    .createTextOutput(JSON.stringify(data))
    .setMimeType(ContentService.MimeType.JSON);
  return output;
}

// Handle preflight OPTIONS request
function doOptions(e) {
  return makeResponse({ ok: true });
}

// Handle GET (health check)
function doGet(e) {
  return makeResponse({ ok: true, message: 'AME webhook active', version: '2.1' });
}

// Handle POST — all write actions
function doPost(e) {
  try {
    const raw = e.postData.contents;
    const data = JSON.parse(raw);
    const ss = SpreadsheetApp.openById(SHEET_ID);
    let result = { ok: true };

    if (data.action === 'submit_artist') {
      const sheet = ss.getSheetByName('submissions_artists');
      sheet.appendRow([
        data.name, data.spotify_id, data.spotify_name,
        data.note, data.submitter_name, data.submitter_email,
        data.timestamp, 'pending'
      ]);

    } else if (data.action === 'submit_playlist') {
      const sheet = ss.getSheetByName('submissions_playlists');
      sheet.appendRow([
        data.spotify_id, data.description, data.about || '',
        data.genre, data.submitter_name, data.submitter_email,
        data.submitter_social || '', data.timestamp, 'pending'
      ]);

    } else if (data.action === 'approve_artist_submission') {
      const subSheet = ss.getSheetByName('submissions_artists');
      const row = subSheet.getRange(data.row + 2, 1, 1, 8).getValues()[0];
      subSheet.getRange(data.row + 2, 8).setValue('approved');
      const artSheet = ss.getSheetByName('artists');
      artSheet.appendRow([row[0], row[1], '', '', '', 'approved', 'manual']);

    } else if (data.action === 'approve_playlist_submission') {
      const subSheet = ss.getSheetByName('submissions_playlists');
      // submissions_playlists columns (0-indexed):
      // 0=spotify_id, 1=description, 2=about, 3=genre, 4=submitter_name,
      // 5=submitter_email, 6=submitter_social, 7=timestamp, 8=status
      const row = subSheet.getRange(data.row + 2, 1, 1, 9).getValues()[0];
      subSheet.getRange(data.row + 2, 9).setValue('approved');
      const plSheet = ss.getSheetByName('playlists');
      // playlists columns: name, spotify_id, genres, curator, description, status, curator_url
      plSheet.appendRow([
        row[1],  // name = description field
        row[0],  // spotify_id
        row[3],  // genres = genre field
        row[4],  // curator = submitter_name
        row[2],  // description = about field
        'approved',
        row[6],  // curator_url = submitter_social
      ]);

    } else if (data.action === 'reject_submission') {
      const sheet = ss.getSheetByName(data.sheet);
      const lastCol = sheet.getLastColumn();
      sheet.getRange(data.row + 2, lastCol).setValue('rejected');

    } else if (data.action === 'add_artist') {
      const sheet = ss.getSheetByName('artists');
      // Column order must match sheet exactly:
      // name | spotify_id | genres | region | bio | status | match_confidence
      sheet.appendRow([
        data.name,
        data.spotify_id,
        data.genres || '',
        data.region || '',
        data.bio || '',
        data.status || 'approved',
        data.match_confidence || 'manual'
      ]);

    } else if (data.action === 'add_playlist') {
      const sheet = ss.getSheetByName('playlists');
      // playlists columns: name, spotify_id, genres, curator, description, status, curator_url
      sheet.appendRow([
        data.name, data.spotify_id, data.genres,
        data.curator || '', data.description || '', data.status,
        data.curator_url || ''
      ]);

    } else if (data.action === 'update_artist_genres') {
      const sheet = ss.getSheetByName('artists');
      // genres is column C (index 3), row is 0-indexed data row so +2 for header
      sheet.getRange(data.row + 2, 3).setValue(data.genres);

    } else if (data.action === 'remove_artist') {
      const sheet = ss.getSheetByName('artists');
      sheet.deleteRow(data.row + 2);

    } else if (data.action === 'remove_playlist') {
      const sheet = ss.getSheetByName('playlists');
      sheet.deleteRow(data.row + 2);

    } else {
      result = { ok: false, error: 'Unknown action: ' + data.action };
    }

    return makeResponse(result);

  } catch (err) {
    return makeResponse({ ok: false, error: err.toString() });
  }
}
